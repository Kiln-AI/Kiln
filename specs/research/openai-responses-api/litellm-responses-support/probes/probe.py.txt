}
MODE = {"tool": False}


class H(BaseHTTPRequestHandler):
    def log_message(self, *a):
        pass

    def do_POST(self):
        body = json.loads(self.rfile.read(int(self.headers["Content-Length"])))
        CAPTURED.append((self.path, body))
        if self.path.endswith("/responses"):
            out = RESP_TOOL_JSON if MODE["tool"] else RESP_JSON
        else:
            out = CHAT_JSON
        data = json.dumps(out).encode()
        self.send_response(200)
        self.send_header("Content-Type", "application/json")
        self.send_header("Content-Length", str(len(data)))
        self.end_headers()
        self.wfile.write(data)


srv = HTTPServer(("127.0.0.1", 0), H)
threading.Thread(target=srv.serve_forever, daemon=True).start()
BASE = f"http://127.0.0.1:{srv.server_port}/v1"

TOOLS = [

RF = {
    "type": "json_schema",
    "json_schema": {
        "name": "out",

            "additionalProperties": False,

        "strict": True,
    },
}
MSGS = [{"role": "system", "content": "sys"}, {"role": "user", "content": "hello"}]


def run(label, tool_resp=False, **kw):
    CAPTURED.clear()
    MODE["tool"] = tool_resp
    kw.setdefault("messages", MSGS)
    try:
        r = litellm.completion(api_base=BASE, api_key="sk-x", drop_params=True, **kw)
        err = None
    except Exception as e:
        r, err = None, f"{type(e).__name__}: {str(e)[:300]}"
    print("=" * 100)
    print(label)
    for path, body in CAPTURED:
        b = {k: v for k, v in body.items() if k not in ("input", "messages")}
        print("  PATH:", path)
        print("  BODY(minus input/messages):", json.dumps(b)[:900])
        if "input" in body:
            print("  INPUT:", json.dumps(body["input"])[:600])
    if err:
        print("  ERROR:", err)
    if r is not None:
        m = r.choices[0].message
        print(
            "  n_choices:",
            len(r.choices),
            "finish:",
            [c.finish_reason for c in r.choices],
        )
        print(
            "  content:",
            repr(m.content),
            "| reasoning_content:",
            repr(getattr(m, "reasoning_content", None)),
        )
        print(
            "  tool_calls[0]:",
            [tc.function.name for tc in (m.tool_calls or [])],
            "| other choices tool_calls:",
            [
                [tc.function.name for tc in (c.message.tool_calls or [])]
                for c in r.choices[1:]
            ],
        )
        print("  reasoning_items:", getattr(m, "reasoning_items", None))
        print("  usage:", r.usage.model_dump() if r.usage else None)
        print("  response_cost:", r._hidden_params.get("response_cost"))
    return r
