import asyncio

import litellm
from probe import BASE, CAPTURED, RF


async def main():
    r = await litellm.aresponses(
        model="openai/gpt-5.5",
        input=[{"role": "user", "content": "hi"}],
        instructions="sys",
        reasoning={"effort": "high", "summary": "auto"},
        text={
            "format": {
                "type": "json_schema",
                "name": "out",
                "schema": RF["json_schema"]["schema"],
                "strict": True,
            }
        },
        store=False,
        include=["reasoning.encrypted_content"],
        api_base=BASE,
        api_key="sk-x",
    )
    print(type(r).__mro__[:3])
    print(
        "path/body:",
        CAPTURED[-1][0],
        {k: v for k, v in CAPTURED[-1][1].items() if k != "input"},
    )
    print("output_text:", getattr(r, "output_text", None))
    print("usage:", r.usage)
    print("hidden cost:", r._hidden_params.get("response_cost"))
    print(
        "completion_cost:",
        litellm.completion_cost(completion_response=r, model="gpt-5.5"),
    )
    print("id:", r.id)


asyncio.run(main())
