import datetime
from decimal import Decimal


class Invoice:
    def __init__(self, customer_id: str, issued: datetime.date):
        self.customer_id = customer_id
        self.issued = issued
        self.lines: list[tuple[str, Decimal, int]] = []

    def add_line(self, description: str, unit_price: Decimal, quantity: int) -> None:
        if quantity <= 0:
            raise ValueError("quantity must be positive")
        self.lines.append((description, unit_price, quantity))

    def subtotal(self) -> Decimal:
        return sum((price * qty for _, price, qty in self.lines), Decimal("0"))

    def tax(self, rate: Decimal) -> Decimal:
        return (self.subtotal() * rate).quantize(Decimal("0.01"))

    def total(self, rate: Decimal) -> Decimal:
        return self.subtotal() + self.tax(rate)

    def render(self) -> str:
        rows = [f"{d}: {qty} x {price}" for d, price, qty in self.lines]
        header = f"Invoice for {self.customer_id} ({self.issued.isoformat()})"
        return "\n".join([header, *rows])


def legacy_total(invoice: Invoice) -> Decimal:
    # Kept for the legacy exporter.
    return invoice.total(Decimal("0.2"))


def format_money(amount: Decimal) -> str:
    return f"${amount:,.2f}"


def parse_money(text: str) -> Decimal:
    cleaned = text.replace("$", "").replace(",", "")
    return Decimal(cleaned)
